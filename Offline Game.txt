Javascript:-

function showOfflineGame() {
  document.getElementById('offline-game').style.display = 'block';
  startGame(); // دالة تشغيل اللعبة
}

function hideOfflineGame() {
  document.getElementById('offline-game').style.display = 'none';
  stopGame(); // دالة إيقاف اللعبة لو حبيت
}

window.addEventListener('offline', () => {
    showOfflineGame();
});
  
  window.addEventListener('online', () => {
    hideOfflineGame();
});
